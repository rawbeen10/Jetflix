FUNCTION get_recommendations_for_user(user, limit):
    
    user_movies = get all movies user has interacted with
    
    IF user_movies is empty:
        RETURN popular movies sorted by (rating, views)
    
    similar_users = find all users who interacted with same movies
                    count how many movies they have in common
                    exclude current user
                    filter only users with >= 1 common movie
                    sort by most common movies
                    take top 20
    
    IF similar_users is empty:
        GO TO genre_based_fallback
    
    recommended_movies = get movies that similar_users interacted with
                         exclude movies user already watched
                         score each movie by how many similar users watched it
                         sort by (score, rating, views)
                         take top [limit]
    
    IF recommended_movies < limit:
        GO TO genre_based_fallback to fill remaining slots
    
    RETURN recommended_movies

    ---------------Genre-Based-------------------------

    FUNCTION genre_based_recommendations(user, user_movies, exclude_ids, limit):
    
    user_genres = get all genres from user's watched movies
    
    candidates = get all published movies
                 where movie has at least one genre in user_genres
                 exclude movies user already watched
                 exclude movies already in recommended list
    
    sort candidates by (rating DESC, views DESC)
    
    RETURN top [limit] candidates
