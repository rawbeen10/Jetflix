1. JACCARD SIMILARITY  (Collaborative Filtering)
────────────────────────────────────────────────────────────────────────────────

FUNCTION get_recommendations(target_user):

    user_movies = movies target_user has watched

    IF user_movies is empty:
        RETURN popular movies sorted by rating, views

    FOR each other_user:
        common  = movies watched by BOTH users
        jaccard = |common| / |user_movies ∪ other_movies|
        IF jaccard > 0: add to similar_users

    similar_users = SORT by jaccard DESC, TAKE top 20

    FOR each similar_user:
        FOR each movie they watched NOT in user_movies:
            candidate_score[movie] += 1

    RETURN top movies by candidate_score DESC


────────────────────────────────────────────────────────────────────────────────
2. GENRE-BASED FILTERING  (Fallback + Search)
────────────────────────────────────────────────────────────────────────────────

FUNCTION genre_filter(user_movies, selected_genres):

    user_genres = all genres from user_movies

    FOR each published movie NOT already watched:
        overlap = |movie.genres ∩ user_genres|
        IF overlap > 0: add to results

    RETURN results sorted by overlap DESC, rating DESC
